{{- $lang := .Type | default "" -}} {{/* Rebuild the options string from the fenced block attributes e.g. ```nim {lineNos=table} becomes "lineNos=table" */}} {{- $opts := slice -}} {{- range $k, $v := .Options -}} {{- $opts = $opts | append (printf "%s=%v" $k $v) -}} {{- end -}} {{- $optsStr := delimit $opts "," -}} {{- $highlighted := transform.Highlight .Inner $lang $optsStr -}}
{{- $highlighted -}}
